Commands to rebuild data, tokenizers, metrics, and smoke training. Prefer Python 3.12 for spaCy wheels. Full H200 curricula are optional.

Environment

python3.12 -m venv .venv
source .venv/bin/activate
make install-dev          # requirements + ruff + spaCy model
make lint
make test

Phase 1 -- assignment data

make verify-ocr            # gate: preprocessed must be Tesseract OCR (exit 1 on issues)
make preprocess
# OCR -> EN join -> HI line-join -> segment -> align -> train/dev/test
# Expect ~1300 pairs (0.6 LaBSE floor), 1010/122/168 split (1458->1422->1300)

Phase 1b -- Stage A + dual policy

make external-download    # if raw files missing
make external-ingest
make external-eval-split
PYTHONPATH=. python -m src.evaluation.eval_sets

Phase 2 -- tokenizers

make tokenizer-train-all
make tokenizer-c0
make tokenizer-spm-v2-full-joint
make tokenizer-bench

Phase 3 -- train smokes

make train-nllb-smoke
# full optional H200:
# make train-nllb-A1-h200
# make train-nllb-A2-h200   # via config training_h200_A2

Phase 4 -- evaluation

make zero-shot-nllb
PYTHONPATH=. python -m src.evaluation.zero_shot_nllb \
  --adapters data/runs/.../checkpoints/best_primary \
  --tag A2_best

# Phase 4 harness extras:
# --score-only honors --max-pairs; reports carry CIs + fingerprints
PYTHONPATH=. python -m src.evaluation.zero_shot_nllb \
  --score-only --max-pairs 190 --tag A2_best
# paired-bootstrap difference CI between two systems on a suite
PYTHONPATH=. python -m src.evaluation.zero_shot_nllb \
  --score-only --compare-tags A2_best,B_best
# COMET-22 over all *_hyps.jsonl (cache keyed on hyp-file SHA256 + model)
make comet-score
# legal-entity panel is emitted inside every *_report.json ('entities' block)

Orchestrator

python run_pipeline.py --list
python run_pipeline.py --steps preprocess
python run_pipeline.py --steps all    # data path only (not multi-hour train)
bash scripts/reproduce_all.sh [--skip-downloads]
